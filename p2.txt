pip install aiml
import aiml
kernel = aiml.Kernel()
kernel.learn("std-startup.xml")
kernel.respond("load aiml b")
while True:
 input_text = input(">Human: ")
 response = kernel.respond(input_text)
 print(">Bot: "+response)

std-startup.xml
<aiml version="1.0.1" encoding="UTF-8">
 <category>
 <pattern> LOAD AIML B </pattern>
 <template>
 <learn>basic_chat.aiml</learn>
 </template>
 </category>
</aiml>
